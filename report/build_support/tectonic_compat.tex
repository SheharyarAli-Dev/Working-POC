  \def\ASENT@natbib{natbib}%
  \ifx\ASENT@package\ASENT@natbib
    \ASENT@originalRequirePackage[sort,numbers]{natbib}%
  \else
    \ASENT@originalRequirePackage[#1]{#2}%
  \fi}
